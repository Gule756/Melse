Core Database Schema Definition (SQL)
-- Enums
CREATE TYPE user_role AS ENUM ('CUSTOMER', 'TECHNICIAN', 'ADMIN', 'BUSINESS');
CREATE TYPE verification_status AS ENUM ('PENDING', 'UNDER_REVIEW', 'VERIFIED', 'SUSPENDED', 'REJECTED');
CREATE TYPE job_status AS ENUM (
  'REQUESTED', 'SEARCHING', 'ASSIGNED', 'ACCEPTED', 
  'TECHNICIAN_EN_ROUTE', 'ARRIVED', 'IN_PROGRESS', 
  'COMPLETED', 'CUSTOMER_CONFIRMED', 'PAID', 'RATED', 
  'CANCELLED', 'DISPUTED', 'REFUNDED'
);
CREATE TYPE payment_status AS ENUM ('PENDING', 'PROCESSING', 'COMPLETED', 'FAILED', 'REFUNDED');

-- Users & Auth
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  phone_number VARCHAR(20) UNIQUE NOT NULL,
  full_name VARCHAR(100) NOT NULL,
  role user_role NOT NULL DEFAULT 'CUSTOMER',
  avatar_url TEXT,
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Services & Pricing
CREATE TABLE service_categories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(50) NOT NULL,
  slug VARCHAR(50) UNIQUE NOT NULL,
  icon_name VARCHAR(50) NOT NULL,
  description TEXT,
  is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE pricing_rules (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  category_id UUID REFERENCES service_categories(id) ON DELETE CASCADE,
  base_price_min DECIMAL(10,2) NOT NULL, -- ETB
  base_price_max DECIMAL(10,2) NOT NULL, -- ETB
  emergency_fee DECIMAL(10,2) DEFAULT 0.00,
  platform_commission_rate DECIMAL(5,2) DEFAULT 0.15, -- 15%
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Technician Profiles
CREATE TABLE technician_profiles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID UNIQUE REFERENCES users(id) ON DELETE CASCADE,
  verification_status verification_status DEFAULT 'PENDING',
  id_document_url TEXT,
  experience_years INT DEFAULT 0,
  rating_avg DECIMAL(3,2) DEFAULT 5.00,
  rating_count INT DEFAULT 0,
  is_available BOOLEAN DEFAULT FALSE,
  current_latitude DECIMAL(10,8),
  current_longitude DECIMAL(11,8),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE technician_skills (
  technician_id UUID REFERENCES technician_profiles(id) ON DELETE CASCADE,
  category_id UUID REFERENCES service_categories(id) ON DELETE CASCADE,
  PRIMARY KEY (technician_id, category_id)
);

-- Customer Addresses
CREATE TABLE customer_addresses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  customer_id UUID REFERENCES users(id) ON DELETE CASCADE,
  label VARCHAR(50), -- e.g. "Home", "Bole Office"
  address_line TEXT NOT NULL,
  subcity VARCHAR(50),
  latitude DECIMAL(10,8) NOT NULL,
  longitude DECIMAL(11,8) NOT NULL,
  is_default BOOLEAN DEFAULT FALSE
);

-- Bookings & Workflow
CREATE TABLE service_requests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  customer_id UUID REFERENCES users(id) ON DELETE RESTRICT,
  technician_id UUID REFERENCES technician_profiles(id) ON DELETE SET NULL,
  category_id UUID REFERENCES service_categories(id) ON DELETE RESTRICT,
  status job_status NOT NULL DEFAULT 'REQUESTED',
  problem_description TEXT NOT NULL,
  problem_photos TEXT[], -- Array of image URLs
  latitude DECIMAL(10,8) NOT NULL,
  longitude DECIMAL(11,8) NOT NULL,
  estimated_price_min DECIMAL(10,2) NOT NULL,
  estimated_price_max DECIMAL(10,2) NOT NULL,
  final_price DECIMAL(10,2),
  commission_amount DECIMAL(10,2),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE booking_status_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id UUID REFERENCES service_requests(id) ON DELETE CASCADE,
  status job_status NOT NULL,
  changed_by UUID REFERENCES users(id),
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Payments & Invoices
CREATE TABLE payments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id UUID UNIQUE REFERENCES service_requests(id) ON DELETE RESTRICT,
  amount DECIMAL(10,2) NOT NULL, -- ETB
  provider VARCHAR(50) NOT NULL, -- Telebirr, Chapa, CBE Birr
  provider_reference VARCHAR(100),
  status payment_status DEFAULT 'PENDING',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Reviews & Guarantees
CREATE TABLE reviews (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id UUID UNIQUE REFERENCES service_requests(id) ON DELETE CASCADE,
  customer_id UUID REFERENCES users(id),
  technician_id UUID REFERENCES technician_profiles(id),
  rating INT CHECK (rating BETWEEN 1 AND 5),
  comment TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE service_guarantees (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id UUID UNIQUE REFERENCES service_requests(id) ON DELETE RESTRICT,
  guarantee_days INT DEFAULT 7,
  valid_until TIMESTAMPTZ NOT NULL,
  status VARCHAR(20) DEFAULT 'ACTIVE' -- ACTIVE, CLAIMED, EXPIRED
);

5. Centralized Job State Machine
To prevent fragmented status strings across the frontend and backend, the job state machine enforces explicit transitions.
// src/constants/jobState.ts

export enum JobStatus {
  REQUESTED = 'REQUESTED',
  SEARCHING = 'SEARCHING',
  ASSIGNED = 'ASSIGNED',
  ACCEPTED = 'ACCEPTED',
  TECHNICIAN_EN_ROUTE = 'TECHNICIAN_EN_ROUTE',
  ARRIVED = 'ARRIVED',
  IN_PROGRESS = 'IN_PROGRESS',
  COMPLETED = 'COMPLETED',
  CUSTOMER_CONFIRMED = 'CUSTOMER_CONFIRMED',
  PAID = 'PAID',
  RATED = 'RATED',
  CANCELLED = 'CANCELLED',
  DISPUTED = 'DISPUTED',
}

export const AllowedStateTransitions: Record<JobStatus, JobStatus[]> = {
  [JobStatus.REQUESTED]: [JobStatus.SEARCHING, JobStatus.CANCELLED],
  [JobStatus.SEARCHING]: [JobStatus.ASSIGNED, JobStatus.CANCELLED],
  [JobStatus.ASSIGNED]: [JobStatus.ACCEPTED, JobStatus.SEARCHING, JobStatus.CANCELLED],
  [JobStatus.ACCEPTED]: [JobStatus.TECHNICIAN_EN_ROUTE, JobStatus.CANCELLED],
  [JobStatus.TECHNICIAN_EN_ROUTE]: [JobStatus.ARRIVED, JobStatus.CANCELLED],
  [JobStatus.ARRIVED]: [JobStatus.IN_PROGRESS, JobStatus.CANCELLED],
  [JobStatus.IN_PROGRESS]: [JobStatus.COMPLETED, JobStatus.DISPUTED],
  [JobStatus.COMPLETED]: [JobStatus.CUSTOMER_CONFIRMED, JobStatus.DISPUTED],
  [JobStatus.CUSTOMER_CONFIRMED]: [JobStatus.PAID],
  [JobStatus.PAID]: [JobStatus.RATED],
  [JobStatus.RATED]: [],
  [JobStatus.CANCELLED]: [],
  [JobStatus.DISPUTED]: [JobStatus.PAID, JobStatus.CANCELLED],
};

export function canTransition(current: JobStatus, next: JobStatus): boolean {
  return AllowedStateTransitions[current]?.includes(next) ?? false;
}

Dynamic Pricing Engine Interface
// src/services/pricing/PricingEngine.ts

export interface PriceCalculationParams {
  categoryId: string;
  isEmergency: boolean;
  distanceKm: number;
  additionalMaterialsFee?: number;
}

export interface EstimatedPriceResult {
  minPrice: number;
  maxPrice: number;
  baseMin: number;
  baseMax: number;
  emergencyFee: number;
  distanceFee: number;
  platformCommission: number; // For internal display/ledger
  currency: 'ETB';
}

export class PricingEngine {
  public static calculateEstimate(
    rules: { baseMin: number; baseMax: number; emergencyFee: number; commissionRate: number },
    params: PriceCalculationParams
  ): EstimatedPriceResult {
    const distanceFee = params.distanceKm > 5 ? (params.distanceKm - 5) * 30 : 0; // 30 ETB per km beyond 5km
    const emergencyAddon = params.isEmergency ? rules.emergencyFee : 0;

    const minPrice = rules.baseMin + emergencyAddon + distanceFee;
    const maxPrice = rules.baseMax + emergencyAddon + distanceFee;

    return {
      minPrice,
      maxPrice,
      baseMin: rules.baseMin,
      baseMax: rules.baseMax,
      emergencyFee: emergencyAddon,
      distanceFee,
      platformCommission: Math.round(minPrice * rules.commissionRate),
      currency: 'ETB',
    };
  }
}
